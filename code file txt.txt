
import React, { useState } from 'react';
import { ICONS } from '../constants';

interface LayoutProps {
  children: React.ReactNode;
  activeTab: string;
  setActiveTab: (tab: string) => void;
}

const Layout: React.FC<LayoutProps> = ({ children, activeTab, setActiveTab }) => {
  const [showPlanEdit, setShowPlanEdit] = useState(false);
  const [planContent, setPlanContent] = useState('Standard Project Management Plan v1.0. High-level objectives and methodology included.');

  const navItems = [
    { id: 'dashboard', label: 'Dashboard', icon: ICONS.DASHBOARD },
    { id: 'stakeholders', label: '5.4 Stakeholder Engagement', icon: ICONS.STAKEHOLDER },
    { id: 'communications', label: '5.5 Communications', icon: ICONS.COMMUNICATION },
    { id: 'logs', label: 'Logs & Outputs', icon: ICONS.LOG },
  ];

  return (
    <div className="flex h-screen bg-slate-50">
      {/* Sidebar */}
      <aside className="w-72 bg-slate-900 text-white flex flex-col shadow-xl">
        <div className="p-6 border-b border-slate-800">
          <h1 className="text-xl font-bold tracking-tight text-blue-400">PMBOK BRIDGE</h1>
          <p className="text-xs text-slate-400 uppercase mt-1 font-semibold">Project Management Hub</p>
        </div>
        
        <nav className="flex-1 mt-6 px-4 space-y-2">
          {navItems.map((item) => (
            <button
              key={item.id}
              onClick={() => setActiveTab(item.id)}
              className={`w-full flex items-center gap-3 px-4 py-3 rounded-lg transition-all ${
                activeTab === item.id 
                  ? 'bg-blue-600 text-white' 
                  : 'text-slate-400 hover:bg-slate-800 hover:text-white'
              }`}
            >
              <span className="text-lg">{item.icon}</span>
              <span className="font-medium">{item.label}</span>
            </button>
          ))}
        </nav>

        <div className="p-6 border-t border-slate-800 bg-slate-950/50">
          <div className="flex items-center gap-3">
            <div className="w-8 h-8 rounded-full bg-blue-500 flex items-center justify-center text-xs font-bold">PM</div>
            <div>
              <p className="text-sm font-medium">Project Manager</p>
              <p className="text-xs text-slate-500">Lead Enterprise</p>
            </div>
          </div>
        </div>
      </aside>

      {/* Main Content */}
      <main className="flex-1 overflow-y-auto">
        <header className="sticky top-0 z-10 bg-white border-b border-slate-200 px-8 py-4 flex justify-between items-center shadow-sm">
          <h2 className="text-xl font-bold text-slate-800">
            {navItems.find(i => i.id === activeTab)?.label}
          </h2>
          <div className="flex gap-4">
             <button 
                onClick={() => setShowPlanEdit(!showPlanEdit)}
                className="px-3 py-1 text-xs border rounded-full hover:bg-slate-50 transition-colors flex items-center gap-1"
             >
                <i className="fa-solid fa-file-lines"></i> {showPlanEdit ? 'Close Plan' : 'View Plan'}
             </button>
             <button className="px-3 py-1 text-xs border rounded-full hover:bg-slate-50 transition-colors flex items-center gap-1">
                <i className="fa-solid fa-shield-halved"></i> Risk Register
             </button>
          </div>
        </header>
        
        {showPlanEdit && (
          <div className="mx-8 mt-4 p-6 bg-blue-50 border border-blue-100 rounded-xl animate-in slide-in-from-top-2 duration-300">
             <h4 className="font-bold text-blue-900 mb-2 flex justify-between items-center">
                Project Management Plan Editor
                <span className="text-[10px] text-blue-400 font-normal">Auto-saving...</span>
             </h4>
             <textarea 
                className="w-full h-24 p-3 rounded-lg border border-blue-200 bg-white text-sm text-slate-700 focus:outline-none focus:ring-2 focus:ring-blue-500"
                value={planContent}
                onChange={(e) => setPlanContent(e.target.value)}
             />
          </div>
        )}

        <div className="p-8 max-w-7xl mx-auto">
          {children}
        </div>
      </main>
    </div>
  );
};

export default Layout;
